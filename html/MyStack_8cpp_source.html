272 unsigned long long hashValue = 0;
273 hashValue += ptrStack->capacity;
274 hashValue *= hashPow;
275 hashValue %= hashMod;
276
277 hashValue += ptrStack->size;
278 hashValue *= hashPow;
279 hashValue %= hashMod;
280
281 for (size_t i = 0; i < ptrStack->capacity; i++) {
282 hashValue += (unsigned long long)ptrStack->data[i];
283 hashValue *= hashPow;
284 hashValue %= hashMod;
285 }
286
287 char* statusString = (char*)ptrStack->status;
288 for (; *statusString != '\0'; statusString++) {
289 hashValue += (unsigned long long)*statusString;
290 hashValue *= hashPow;
